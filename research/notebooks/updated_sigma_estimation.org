* Basic idea
The estimation method used by [cite:@horneAnalyzingAnimalMovements2007] doesn't take into account measurement errors in the right way, as shown by [cite:@pozdnyakovModelingAnimalMovements2014]. I now try to implement the newer version of parameter estimation proposed by the latter.

Let us first consider the current implementation. It uses odd positions as target points and tries to maximize the log likelihood of the variance based on that. This is done by the following metho (written in a slightly different way to incorporate csv data instead of geometric df).

For all the experiments, we will use the data used in the example of [cite:@pozdnyakovModelingAnimalMovements2014] about the locations of a female mule deer.

#+name: current_impl
#+begin_src jupyter-python
import numpy as np
import pandas as pd
from scipy.optimize import minimize_scalar

def L_mod(gdf, sigma_m, delta_z):
    result = 0
    for i in range(0, len(gdf) - 2, 2):
        start = gdf.iloc[i]
        mid = gdf.iloc[i + 1]
        end = gdf.iloc[i + 2]
        T_i = (end["time.lag"] - start["time.lag"])
        alpha = (mid["time.lag"] - start["time.lag"]) / T_i
        mu_tx = start["x"] + alpha * (end["x"] - start["x"])
        mu_ty = start["y"] + alpha * (end["y"] - start["y"])
        sigma_t = (
            T_i * alpha * (1 - alpha) * sigma_m**2
            + ((1 - alpha) ** 2 + alpha ** 2) * delta_z ** 2
        )
        result += -np.log(2 * np.pi * sigma_t) - (
            ((mid["x"] - mu_tx) ** 2 + (mid["y"] - mu_ty) ** 2)
            / (2 * sigma_t)
        )

    return result

def calculate_sigma_m(gdf, delta_z):
    result = minimize_scalar(lambda sigma_m: -L_mod(gdf, sigma_m, delta_z))
    sigma_m = result.x
    return sigma_m

I now implement the updated version of this estimation. We will start by creating a sparse banded matrix for the sigma values. This can be done using the =scipy.sparse.dia_matrix= function.

#+begin_src jupyter-python
from scipy.sparse import dia_matrix

def get_sparse_sigma(time: np.ndarray, parameter: np.ndarray):
    n = len(time)
    tau = np.ediff1d(time)
    d0 = tau * parameter[0]**2 + 2 * parameter[1]**2
    d1 = np.ones(n-1) * -parameter[1]**2
    offsets = np.array([-1, 0, 1])
    data = np.array([d1, d0, d1])
    sigma_diag = dia_matrix((data, offsets), shape=(n-1, n-1)).toarray()
    return sigma_diag

The above covariance matrix can now be used to calculate the multivariate density.

#+name: multivariate_density
#+begin_src jupyter-python
import scipy.linalg

def multivariate_density(x: np.ndarray, cov_mat):
    n = cov_mat.shape[0]
    c, lower = scipy.linalg.cho_factor(cov_mat)
    p1 = - (n / 2) * np.log(2 * np.pi) - np.log(np.diag(c)).sum()
    y = scipy.linalg.cho_solve((c, lower), x)
    p2 = - (np.dot(x, y))/2
    return p1 + p2
#+end_src

#+RESULTS: multivariate_density

This multivariate density can now be used to implement negative loglikelihood estimation and estimate the values of both, variance \sigma_m^2 and measurement_error \delta_m

#+name: loglikelihood
#+begin_src jupyter-python
from scipy.optimize import minimize

def negative_loglikelihood(parameters: np.ndarray, df: pd.DataFrame):
    zxi_s = np.ediff1d(df['x'])
    sigma_x = get_sparse_sigma(df['time.lag'], parameters)
    loglike_x = multivariate_density(zxi_s, sigma_x)

    zyi_s = np.ediff1d(df['y'])
    sigma_y = get_sparse_sigma(df['time.lag'], parameters)
    loglike_y = multivariate_density(zyi_s, sigma_y)

    likelihood = -(loglike_x + loglike_y)
    # print(loglike_x)
    return likelihood

def calculate_parameters(df):
    n = df.shape[0]
    tau = np.ediff1d(df['time.lag'])
    zxi_s = np.ediff1d(df['x'])
    zyi_s = np.ediff1d(df['y'])
    start_val = np.sqrt((zxi_s**2 + zyi_s**2).sum()/(2 * (2 + tau).sum()))
    result = minimize(negative_loglikelihood, [start_val]*2, args=(df,))
    return result.x

rng = np.random.default_rng()
def gen_data(time):
    tau = np.ediff1d(time)
    s = rng.uniform(low=40, high=60)
    std_dev = np.sqrt(tau * s**2)

    true_x = np.concatenate([[0], np.cumsum(rng.normal(0, std_dev))])
    true_y = np.concatenate([[0], np.cumsum(rng.normal(0, std_dev))])

    error_x = delta_z * rng.standard_normal(size=time.shape[0])
    error_y = delta_z * rng.standard_normal(size=time.shape[0])

    x = true_x + error_x
    y = true_y + error_y
    df = pd.DataFrame({'time.lag': time, 'x': x, 'y': y})
    return df, s

def generate_random_data(delta_z):
    time = np.arange(0, 1000, 5)
    indices = np.random.default_rng().integers(0, len(time)-1, size=int(0.3*len(time)))
    time_2 = np.delete(time, indices)

    df, s = gen_data(time)
    # df_2, s_2 = gen_data(time_2)
    df_2 = df.iloc[df.index.delete(indices)]
    df_2 = df_2.reset_index()
    return df, df_2, s, s
